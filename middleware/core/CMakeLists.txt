# WP-AL: DTV middleware library dtvcore (AL modules) and its unit tests (TST-1xx).
# Portability rule R3 / NFR-09: dtvcore uses only the C++17 standard library, nlohmann/json and
# the IF-INT-04 headers; dtvcore_tests links only dtvcore and GoogleTest (TST-159).
add_library(dtvcore STATIC
    psi.cpp
    task_queue.cpp
)
target_include_directories(dtvcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dtvcore PUBLIC pal_api nlohmann_json)

add_executable(dtvcore_tests
    tests/fakes/fake_demux.cpp
    tests/fakes/fake_hls_store.cpp
    tests/fakes/fake_http_client.cpp
    tests/fakes/reference_d04.cpp
    tests/fakes_platform_test.cpp
    tests/fakes_test.cpp
    tests/psi_builder.cpp
    tests/psi_test.cpp
    tests/task_queue_test.cpp
    tests/ts_builder.cpp
)
target_include_directories(dtvcore_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
# Content files of Section 5.2.1 (not committed; generated by content/generate.sh).
target_compile_definitions(dtvcore_tests PRIVATE
    M15_CONTENT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../content"
    # Default responses of the fake HTTP client (JSON examples of Section 3; checked by TST-533).
    M15_FAKE_RESPONSES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fakes/responses")
target_link_libraries(dtvcore_tests PRIVATE dtvcore GTest::gtest_main)
gtest_discover_tests(dtvcore_tests)
